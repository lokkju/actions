# AI attribution in commit messages, PR titles and PR bodies.
#
# One POSIX extended regular expression per line, matched case-insensitively against
# each line of text on its own (GNU grep -Ei, LC_ALL=C; \b is GNU's word boundary, so
# a BSD grep needs ggrep). Lines starting with # and blank
# lines are ignored. A blank pattern would match everything, so the scanner strips them.
#
# Match trailer lines, known boilerplate and session links, never a bare tool name: a PR
# that discusses a tool or an environment variable such as OPENAI_API_KEY must pass.
# Every pattern needs a positive case in tests/positive.txt, and every prose shape that
# a pattern could plausibly hit needs a case in tests/negative.txt.

# --- Co-author trailers ----------------------------------------------------------------

# A co-author whose name carries an unambiguous AI tool or vendor name.
^[[:space:]]*co-authored-by:[^<]*\b(chatgpt|openai|gpt-?[0-9]|copilot|gemini|codeium|windsurf|aider|tabnine|codewhisperer|amazon q|sourcegraph|deepseek|qwen|grok|cline|roo ?code|kilo ?code|coderabbit|augment ?code|opencode|anthropic|cognition|lovable|claude code)
# A co-author named only after a tool that is also a human first name (Claude, Devin,
# Jules, ...): the whole name is the tool, optionally followed by a model or "AI".
^[[:space:]]*co-authored-by:[[:space:]]*(claude|devin|jules|cursor|cody|junie|amp|codex)([[:space:]]+(opus|sonnet|haiku|fable|instant|agent|ai|code|cli|[0-9][^<[:space:]]*))*[[:space:]]*(<|$)
# A co-author that is an AI tool's GitHub App or bot account.
^[[:space:]]*co-authored-by:.*[+<[:space:]](claude|devin-ai-integration|google-labs-jules|copilot-swe-agent|copilot|cursoragent|cursor-agent|cursor|gemini-code-assist|chatgpt-codex-connector|codex|jules|junie|coderabbitai|sweep-ai|amp)(\[bot\]|@users\.noreply\.github\.com)
# A co-author at an AI vendor's domain.
^[[:space:]]*co-authored-by:.*@([a-z0-9-]+\.)?(anthropic\.com|openai\.com|cursor\.(com|sh)|anysphere\.co|codeium\.com|windsurf\.com|cognition\.ai|cognition-labs\.com|devin\.ai|aider\.chat|tabnine\.com|coderabbit\.ai|augmentcode\.com|sweep\.dev|cline\.bot|kilocode\.ai|ampcode\.com|opencode\.ai|x\.ai|deepseek\.com|mistral\.ai|lovable\.dev)>?

# --- Vendor noreply and bot addresses on any trailer --------------------------------------

^[[:space:]]*[a-z][a-z-]*:.*(noreply|no-reply|bot|agent)[^@[:space:]<]*@([a-z0-9-]+\.)?(anthropic\.com|openai\.com|cursor\.(com|sh)|anysphere\.co|codeium\.com|windsurf\.com|cognition\.ai|devin\.ai|aider\.chat|tabnine\.com|coderabbit\.ai|augmentcode\.com|sweep\.dev|ampcode\.com|opencode\.ai|x\.ai|deepseek\.com|mistral\.ai)

# --- Provenance and session trailers ------------------------------------------------------

# Generated-by, Assisted-by and their siblings, whatever tool they name.
^[[:space:]]*(generated|assisted|created|written|authored|co-developed|coded|made|produced|drafted)[- ](by|with|using)[[:space:]]*:
# AI-Assisted:, LLM-Model:, AI-Tool:, Agent-Session: and similar.
^[[:space:]]*(ai|llm|genai|agent|model)[- ](assisted|generated|tool|tools|model|session|agent|provenance|usage|disclosure)[[:space:]]*:
# X-AI-...: style headers used as trailers.
^[[:space:]]*x-(ai|llm|genai)(-[a-z]+)*[[:space:]]*:
# The session trailer a coding harness appends (Claude-Session:, Codex-Session:, ...).
^[[:space:]]*(claude|codex|cursor|copilot|gemini|aider|opencode|devin|jules|amp|ai)[- ]session(-id|-url|-link)?[[:space:]]*:
